> # -----------------------------------
> # Двухвыборочный t-критерий
> # Вариант 1
> # -----------------------------------
> 
> # Рост эльфов
> elves <- c(185, 188, 182, 182, 184,
+            186, 183, 181, 176, 180,
+            178, 189, 186, 170, 183)
> 
> # Рост неизвестных фигур
> figures <- c(162, 158, 161, 166, 164,
+              159, 172, 170, 163, 166,
+              166, 164, 169, 165, 167,
+              170, 176)
> 
> # -----------------------------------
> # Формулировка гипотез
> # -----------------------------------
> 
> # H0 (нулевая гипотеза):
> # Средний рост неизвестных фигур
> # не отличается от среднего роста эльфов.
> 
> # H1 (альтернативная гипотеза):
> # Средний рост неизвестных фигур
> # отличается от среднего роста эльфов.
> 
> # -----------------------------------
> # Описательная статистика
> # -----------------------------------
> 
> mean(elves)
[1] 182.2
> sd(elves)
[1] 4.872664
> 
> mean(figures)
[1] 165.7647
> sd(figures)
[1] 4.670779
> 
> # -----------------------------------
> # Проверка гипотезы
> # -----------------------------------
> 
> t.test(elves, figures)

        Welch Two Sample t-test

data:  elves and figures
t = 9.7079, df = 29.142, p-value = 1.224e-10
alternative hypothesis: true difference in means is not equal to 0
95 percent confidence interval:
 12.97351 19.89708
sample estimates:
mean of x mean of y 
 182.2000  165.7647 

> 
> # -----------------------------------
> # Интерпретация результата
> # -----------------------------------
> 
> # Если p-value < 0.05,
> # нулевая гипотеза отвергается.
> 
> # В данном случае p-value значительно меньше 0.05,
> # поэтому можно сделать вывод,
> # что средний рост неизвестных фигур
> # статистически значимо отличается
> # от среднего роста эльфов.
> 
> # Следовательно,
> # обнаруженные существа не являются эльфами.
> 
> # -----------------------------------
> # Построение столбчатой диаграммы
> # -----------------------------------
> 
> means <- c(mean(elves), mean(figures))
> 
> barplot(means,
+         names.arg = c("Эльфы", "Неизвестные"),
+         col = c("lightblue", "lightgreen"),
+         main = "Сравнение среднего роста",
+         ylab = "Рост (см)")
